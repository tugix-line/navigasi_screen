import 'package:flutter/material.dart';

class ProdukPage extends StatelessWidget {
  const ProdukPage({super.key});

  // Data daftar produk
  static const List<Product> daftarProduk = [
    Product('Laptop ASUS', 'Elektronik', 'Rp 8.500.000'),
    Product('Samsung Galaxy A55', 'Smartphone', 'Rp 5.500.000'),
    Product('Mouse Logitech', 'Aksesoris Komputer', 'Rp 250.000'),
    Product('Headphone Sony', 'Audio', 'Rp 1.200.000'),
    Product('Monitor LG 24 Inch', 'Elektronik', 'Rp 1.800.000'),
  ];

  // Function warna produk
  Color warnaIconProduk(int index) {
    List<Color> warna = [
      Colors.blue,
      Colors.green,
      Colors.orange,
      Colors.purple,
      Colors.red,
      Colors.teal,
      Colors.pink,
      Colors.indigo,
    ];
    return warna[index % warna.length];
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(
          leading: IconButton(
            onPressed: () {
              Navigator.pushReplacement(
                context,
                MaterialPageRoute(builder: (context) => MainApp()),
              );
            },
            icon: Icon(Icons.arrow_back),
          ),
          title: Text('Daftar Produk'),
          centerTitle: true,
        ),
        body: ListView.builder(
          padding: const EdgeInsets.all(12),
          itemCount: daftarProduk.length,
          itemBuilder: (context, index) {
            final produk = daftarProduk[index];
            return Card(
              elevation: 2,
              margin: const EdgeInsets.only(bottom: 10),
              child: ListTile(
                leading: CircleAvatar(
                  backgroundColor: warnaIconProduk(index),
                  child: Icon(Icons.inventory_2, color: Colors.white),
                ),
                title: Text(
                  produk.nama,
                  style: const TextStyle(
                    fontWeight: FontWeight.bold,
                    fontSize: 16,
                  ),
                ),
                subtitle: Text(
                  produk.kategori,
                  style: const TextStyle(color: Colors.grey),
                ),
                trailing: const Icon(Icons.arrow_forward_ios, size: 18),
                // Navigasi menuju halaman detail produk
                onTap: () {
                  Navigator.push(
                    context,
                    MaterialPageRoute(
                      builder: (context) => DetailProdukPage(produk: produk),
                    ),
                  );
                },
              ),
            );
          },
        ),
      ),
    );
  }
}
